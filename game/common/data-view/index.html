组件调用示例
JavaScript 调用方式
// 引入 DataViewer3D 组件
import { DataViewer3D } from './common/data-view/viewer.js';

// 创建可视化实例
const viewer = new DataViewer3D({
    container: '#canvas-container',
    type: 'molecule',  // molecule | data | space
    data: moleculeData,
    color: '#00d4aa',
    background: '#1a2a4a'
});

// 渲染
viewer.render();

// 更新数据
viewer.update(newData);

// 销毁
viewer.dispose();
